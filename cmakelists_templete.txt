cmake_minimum_required(VERSION 3.16)
project(sylar LANGUAGES CXX)

# ====== 编译选项 ======
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# 可选：显示详细编译信息
set(CMAKE_VERBOSE_MAKEFILE ON)

# ====== 输出目录统一 ======
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${PROJECT_SOURCE_DIR}/bin)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${PROJECT_SOURCE_DIR}/lib)
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY ${PROJECT_SOURCE_DIR}/lib)

# ====== Boost ======
find_package(Boost REQUIRED)
if(NOT Boost_FOUND)
    message(FATAL_ERROR "Boost not found! Please install Boost or set Boost_DIR")
endif()
# 对 sylar 及依赖 target 生效
include_directories(${Boost_INCLUDE_DIRS})

# ====== yaml-cpp ======
find_package(yaml-cpp REQUIRED)
if(NOT yaml-cpp_FOUND)
    message(FATAL_ERROR "yaml-cpp not found! Please install yaml-cpp or set yaml-cpp_DIR")
endif()

# ====== 项目头文件 ======
# sylar 自己的头文件
target_include_directories(sylar PUBLIC
    ${PROJECT_SOURCE_DIR}
    ${Boost_INCLUDE_DIRS}
    ${YAML_CPP_INCLUDE_DIRS}
)

# ====== 源文件 ======
set(LIB_SRC
    sylar/log.cc
    sylar/util.cc
    sylar/config.cc
)

# ====== 动态库 ======
add_library(sylar SHARED ${LIB_SRC})

# ====== 可执行文件 ======
add_executable(test tests/test.cc)
add_dependencies(test sylar)
target_link_libraries(test PRIVATE sylar ${Boost_LIBRARIES} yaml-cpp)

add_executable(test_config tests/test_config.cc)
add_dependencies(test_config sylar)
target_link_libraries(test_config PRIVATE sylar ${Boost_LIBRARIES} yaml-cpp)

# ====== 生成 compile_commands.json ======
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

# ====== 可选：针对不同配置输出目录 ======
# 多配置生成器（Xcode / Visual Studio）时
foreach(CONFIG_TYPE IN ITEMS Debug Release RelWithDebInfo MinSizeRel)
    set_target_properties(sylar PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${CONFIG_TYPE} ${PROJECT_SOURCE_DIR}/bin)
    set_target_properties(test PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${CONFIG_TYPE} ${PROJECT_SOURCE_DIR}/bin)
    set_target_properties(test_config PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${CONFIG_TYPE} ${PROJECT_SOURCE_DIR}/bin)
    set_target_properties(sylar PROPERTIES LIBRARY_OUTPUT_DIRECTORY_${CONFIG_TYPE} ${PROJECT_SOURCE_DIR}/lib)
endforeach()
